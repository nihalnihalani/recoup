# Recoup Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Ship Recoup, a Convex web app that turns a forwarded order email into item-level records, opens claims for under-credited returns and in-window price drops, sends one approved request per claim through AgentMail, classifies replies, and tracks money until the user confirms it posted.

**Architecture:** One `purchases` case with `items`; a `claims` row per item and trigger (`return_credit` or `price_adjustment`); an append-only `ledgerEvents` table whose sums are derived in queries, never stored. Pure ledger math lives in `convex/lib/ledger.ts` and is unit-tested without Convex. AgentMail, Firecrawl and static hosting are Convex components; OpenAI is called from actions with zod structured outputs and only ever proposes data that the user confirms.

**Tech Stack:** React 19, Vite, TypeScript, Tailwind v4, react-router-dom 7, Convex, `@convex-dev/auth` (Password), `@agentmail/convex` 0.1, `@firecrawl/firecrawl-convex` 0.1, `@convex-dev/static-hosting` 0.2, `openai` (Responses API, `gpt-5.6-terra`), `zod` 3, `vitest` + `convex-test`.

**Step 3: Implement convex/lib/ledger.ts**

```ts
export type EventKind = "promised_credit" | "confirmed_credit" | "later_debit";
export type LedgerEvent = { kind: EventKind; cents: number };
export type ClaimStatus =
  | "detected" | "drafted" | "sent" | "packet"
  | "promised" | "confirmed" | "reopened" | "dismissed";

export type Balance = {
  expected: number;
  promised: number;
  confirmed: number;
  debited: number;
  unresolved: number;
};

export function balance(expectedCents: number, events: LedgerEvent[]): Balance {
  let promised = 0;
  let confirmed = 0;
  let debited = 0;
  for (const e of events) {
    if (!Number.isInteger(e.cents) || e.cents < 0) {
      throw new Error("ledger cents must be a non-negative integer");
    }
    if (e.kind === "promised_credit") promised += e.cents;
    else if (e.kind === "confirmed_credit") confirmed += e.cents;
    else debited += e.cents;
  }
  return {
    expected: expectedCents,
    promised,
    confirmed,
    debited,
    unresolved: expectedCents - confirmed + debited,
  };
}

export function isSettled(b: Balance): boolean {
  return b.confirmed > 0 && b.unresolved <= 0;
}

/** Minimum drop worth a claim: the larger of $1.00 or 2% of the paid price. */
export function priceDropCents(unitCents: number, observedCents: number, qty: number): number | null {
  const drop = unitCents - observedCents;
  const threshold = Math.max(100, Math.round(unitCents * 0.02));
  if (drop < threshold) return null;
  return drop * qty;
}

export function windowEndsAt(purchasedAt: number, windowDays: number): number {
  return purchasedAt + windowDays * 86_400_000;
}

export function statusAfterEvent(current: ClaimStatus, kind: EventKind, b: Balance): ClaimStatus {
  if (current === "dismissed") return current;
  if (kind === "later_debit") return b.unresolved > 0 ? "reopened" : current;
  if (kind === "confirmed_credit") {
    if (isSettled(b)) return "confirmed";
    return current === "confirmed" ? "reopened" : current;
  }
  // promised_credit
  return current === "confirmed" ? current : "promised";
}

export function newToken(): string {
  const alphabet = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  let out = "";
  for (let i = 0; i < 6; i++) out += alphabet[Math.floor(Math.random() * alphabet.length)];
  return out;
}

/** Extracts the claim token from a subject like "Re: Order 123 [RC-AB12CD]". */
export function tokenFromSubject(subject: string | undefined): string | null {
  const m = /\[RC-([A-Z0-9]{6})\]/.exec(subject ?? "");
  return m ? m[1] : null;
}
```

import type { Id } from "../_generated/dataModel";
import type { MutationCtx, QueryCtx } from "../_generated/server";

type Ctx = QueryCtx | MutationCtx;

export async function requireUserId(ctx: Ctx): Promise<Id<"users">> {
  const userId = await getAuthUserId(ctx);
  if (!userId) throw new ConvexError("Not signed in");
  return userId;
}

export async function ownedPurchase(ctx: Ctx, purchaseId: Id<"purchases">, userId: Id<"users">) {
  const p = await ctx.db.get(purchaseId);
  if (!p || p.userId !== userId) throw new ConvexError("Purchase not found");
  return p;
}

  const i = await ctx.db.get(itemId);

export async function openClaim(ctx: MutationCtx, args: {
  userId: Id<"users">; purchaseId: Id<"purchases">; itemId: Id<"items">;
  type: "price_adjustment" | "return_credit"; expectedCents: number;
  windowEndsAt?: number; policyId?: Id<"policies">; openedFromPriceCheckId?: Id<"priceChecks">;
}) {
  if (!Number.isInteger(args.expectedCents) || args.expectedCents <= 0) throw new ConvexError("expectedCents must be a positive integer");
  const existing = await ctx.db.query("claims").withIndex("by_item", (q) => q.eq("itemId", args.itemId)).collect();
  if (existing.some((c) => c.type === args.type && !["confirmed", "dismissed"].includes(c.status))) {
    throw new ConvexError("An open claim of this type already exists for this item");
  }
  return ctx.db.insert("claims", { ...args, status: "detected", token: newToken(), version: 1 });
}

export async function applyEvent(ctx: MutationCtx, claim: Doc<"claims">, kind: EventKind, cents: number, evidence: string, idempotencyKey?: string) {
  if (idempotencyKey) {
    const dup = await ctx.db.query("ledgerEvents").withIndex("by_key", (q) => q.eq("idempotencyKey", idempotencyKey)).first();
    if (dup) return { deduped: true as const, status: claim.status };
  }
  if (!Number.isInteger(cents) || cents < 0) throw new ConvexError("cents must be a non-negative integer");
  await ctx.db.insert("ledgerEvents", { claimId: claim._id, userId: claim.userId, kind, cents, evidence, idempotencyKey });
  const events = await ctx.db.query("ledgerEvents").withIndex("by_claim", (q) => q.eq("claimId", claim._id)).collect();
  const b = balance(claim.expectedCents, events);
  const status = statusAfterEvent(claim.status, kind, b);
  const patch: Partial<Doc<"claims">> = { status, version: claim.version + 1 };
  if (status === "confirmed") { await cancelPending(ctx, claim._id); patch.attentionAt = undefined; }
  await ctx.db.patch(claim._id, patch);
  return { deduped: false as const, status };
}

import { ConvexError, v } from "convex/values";
import { action, internalAction, internalMutation } from "./_generated/server";
import { internal } from "./_generated/api";
import { getAuthUserId } from "@convex-dev/auth/server";

const SYSTEM = `You read a single email a shopper forwarded. Decide whether it is an order confirmation ("order"), a refund or return-credit notice ("refund"), or neither ("other"). Extract only what the email states. Prices are per unit before tax. If the email lists a product link, include it. Never invent a merchant domain: derive it from sender or links.`;

export const extractFromText = internalAction({
  args: { userId: v.id("users"), sourceMessageId: v.optional(v.string()), subject: v.string(), text: v.string(), from: v.string() },
  handler: async (ctx, args) => {
    const parsed = await extract("inbound_email", InboundEmail, SYSTEM, `From: ${args.from}\nSubject: ${args.subject}\n\n${args.text}`);
    await ctx.runMutation(internal.intake.applyExtraction, { userId: args.userId, sourceMessageId: args.sourceMessageId, parsed });
  },

export const applyExtraction = internalMutation({
  args: { userId: v.id("users"), sourceMessageId: v.optional(v.string()), parsed: v.any() },
  handler: async (ctx, { userId, sourceMessageId, parsed }) => {
    const p = InboundEmail.parse(parsed);
    if (p.kind === "order" && p.order) {
      const purchaseId = await ctx.db.insert("purchases", {
        userId, merchant: p.order.merchant, merchantDomain: p.order.merchantDomain.toLowerCase(),
        orderRef: p.order.orderRef ?? undefined, purchasedAt: Date.parse(p.order.purchasedAt) || Date.now(),
        currency: p.order.currency.toUpperCase(), sourceMessageId,
        status: p.confidence >= 0.8 ? "active" : "needs_review",
      });
      for (const it of p.order.items) {
        await ctx.db.insert("items", { purchaseId, userId, name: it.name, unitCents: toCents(it.unitPrice), qty: it.qty, productUrl: it.productUrl ?? undefined, returned: false });
      }
      if (p.confidence >= 0.8) {
        await ctx.scheduler.runAfter(0, internal.policies.fetchBoth, { userId, merchantDomain: p.order.merchantDomain.toLowerCase() });
      }
      return { kind: "order", purchaseId };
    }
    if (p.kind === "refund" && p.refund) {
      // Find the purchase by order ref, else by merchant name, for this user only.
      const purchases = await ctx.db.query("purchases").withIndex("by_user", (q) => q.eq("userId", userId)).collect();
      const purchase = purchases.find((x) => p.refund!.orderRef && x.orderRef === p.refund!.orderRef)
        ?? purchases.find((x) => p.refund!.merchant && x.merchant.toLowerCase().includes(p.refund!.merchant!.toLowerCase()));
      if (!purchase) return { kind: "refund", matched: false };
      const items = await ctx.db.query("items").withIndex("by_purchase", (q) => q.eq("purchaseId", purchase._id)).collect();
      const returnsPolicy = await ctx.db.query("policies")
        .withIndex("by_user_domain_kind", (q) => q.eq("userId", userId).eq("merchantDomain", purchase.merchantDomain).eq("kind", "returns")).first();
      for (const credit of p.refund.credits) {
        const cents = toCents(credit.amount);
        const item = items.find((i) => credit.itemName && i.name.toLowerCase().includes(credit.itemName.toLowerCase()))
          ?? items.find((i) => i.unitCents * i.qty === cents);
        if (!item) continue;
        if (!item.returned) await ctx.db.patch(item._id, { returned: true, returnedAt: Date.now() });
        let claim = (await ctx.db.query("claims").withIndex("by_item", (q) => q.eq("itemId", item._id)).collect())
          .find((c) => c.type === "return_credit" && c.status !== "dismissed");
        if (!claim) {
          const id = await openClaim(ctx, { userId, purchaseId: purchase._id, itemId: item._id, type: "return_credit", expectedCents: item.unitCents * item.qty, policyId: returnsPolicy?._id });
          claim = (await ctx.db.get(id))!;
        }
        await applyEvent(ctx, claim, "promised_credit", cents, `Merchant email says refund ${credit.state} (${sourceMessageId ?? "pasted"})`, sourceMessageId ? `msg:${sourceMessageId}:${item._id}` : undefined);
      }
      // Every other returned item without a credit gets an open claim so the gap is visible.
      for (const item of items) {
        if (!item.returned) continue;
        const has = (await ctx.db.query("claims").withIndex("by_item", (q) => q.eq("itemId", item._id)).collect()).some((c) => c.type === "return_credit" && c.status !== "dismissed");
        if (!has) await openClaim(ctx, { userId, purchaseId: purchase._id, itemId: item._id, type: "return_credit", expectedCents: item.unitCents * item.qty, policyId: returnsPolicy?._id });
      }
      return { kind: "refund", matched: true, purchaseId: purchase._id };
    }
    return { kind: "other" };
  },
});

```ts
import { ConvexError, v } from "convex/values";
import { FirecrawlClient } from "@firecrawl/firecrawl-convex";
import { action, internalAction, internalMutation, mutation } from "./_generated/server";
import { components, internal } from "./_generated/api";
import { getAuthUserId } from "@convex-dev/auth/server";
import { Policy, extract } from "./lib/ai";
import { channel, policyKind } from "./schema";
import { requireUserId } from "./lib/access";

const firecrawl = new FirecrawlClient(components.firecrawl);

const QUERIES = {
  price_adjustment: (d: string) => `${d} price adjustment policy price match after purchase`,
  returns: (d: string) => `${d} return refund policy how long refund takes contact`,
} as const;

const SYSTEM = (kind: "price_adjustment" | "returns") => kind === "price_adjustment"
  ? `You read a retailer's policy page. Determine whether the retailer adjusts the price of an item already purchased when its own price drops. Report the window in days, how to request it (email, form, chat, phone, unknown), a contact email if the page publishes one, and the verbatim passage. Set found=false if the page does not cover price adjustments.`
  : `You read a retailer's returns page. Report how many days after receipt the retailer says a refund is processed or when to contact support (windowDays), the documented channel for refund questions, a published support email if any, and the verbatim passage.`;

export const fetchBoth = internalAction({
  args: { userId: v.id("users"), merchantDomain: v.string() },
  handler: async (ctx, args) => {
    await ctx.runAction(internal.policies.fetchOne, { ...args, kind: "price_adjustment" });
    await ctx.runAction(internal.policies.fetchOne, { ...args, kind: "returns" });
  },
});

export const fetchOne = internalAction({
  args: { userId: v.id("users"), merchantDomain: v.string(), kind: policyKind },
  handler: async (ctx, { userId, merchantDomain, kind }) => {
    const res = await firecrawl.search(ctx, QUERIES[kind](merchantDomain), {
      limit: 3,
      includeDomains: [merchantDomain],
      scrapeOptions: { formats: ["markdown"], onlyMainContent: true, maxAge: 86_400_000 },
    });
    const hits = (res.web ?? []).filter((h: any) => typeof h.markdown === "string" && h.markdown.length > 200);
    if (hits.length === 0) {
      await ctx.runMutation(internal.policies.upsert, { userId, merchantDomain, kind, channel: "unknown", passage: "No policy page found on the merchant's domain.", sourceUrl: `https://${merchantDomain}`, confidence: 0 });
      return;
    }
    let best: any = null;
    for (const h of hits) {
      const parsed = await extract("policy", Policy, SYSTEM(kind), h.markdown as string);
      if (!best || parsed.confidence > best.parsed.confidence) best = { parsed, url: h.url ?? h.metadata?.sourceURL };
      if (parsed.found && parsed.confidence >= 0.8) break;
    }
    const p = best.parsed;
    await ctx.runMutation(internal.policies.upsert, {
      userId, merchantDomain, kind, windowDays: p.windowDays ?? undefined, channel: p.channel,
      contactEmail: p.contactEmail ?? undefined, passage: p.passage.slice(0, 600), sourceUrl: best.url, confidence: p.found ? p.confidence : 0,
    });
  },
});

export const upsert = internalMutation({
  args: { userId: v.id("users"), merchantDomain: v.string(), kind: policyKind, windowDays: v.optional(v.number()), channel, contactEmail: v.optional(v.string()), passage: v.string(), sourceUrl: v.string(), confidence: v.number() },
  handler: async (ctx, args) => {
    const existing = await ctx.db.query("policies")
      .withIndex("by_user_domain_kind", (q) => q.eq("userId", args.userId).eq("merchantDomain", args.merchantDomain).eq("kind", args.kind)).first();
    if (existing?.confirmedByUser) return existing._id;
    const doc = { ...args, retrievedAt: Date.now(), confirmedByUser: false };
    if (existing) { await ctx.db.patch(existing._id, doc); return existing._id; }
    return ctx.db.insert("policies", doc);
  },
});

export const refresh = action({
  args: { merchantDomain: v.string(), kind: policyKind },
  handler: async (ctx, args) => {
    const userId = await getAuthUserId(ctx);
    if (!userId) throw new ConvexError("Not signed in");
    await ctx.runAction(internal.policies.fetchOne, { userId, ...args });
  },
});

export const confirm = mutation({
  args: { policyId: v.id("policies"), windowDays: v.optional(v.number()), channel, contactEmail: v.optional(v.string()), passage: v.optional(v.string()), sourceUrl: v.optional(v.string()) },
  handler: async (ctx, { policyId, ...edits }) => {
    const userId = await requireUserId(ctx);
    const p = await ctx.db.get(policyId);
    if (!p || p.userId !== userId) throw new ConvexError("Policy not found");
    await ctx.db.patch(policyId, { ...edits, confirmedByUser: true });
  },
});
```

    const out: { itemId: typeof items[number]["_id"] }[] = [];

  },

export const runAll = internalAction({
  args: {},
  handler: async (ctx) => {
    const items = await ctx.runQuery(internal.priceWatch.eligibleItems, {});
    for (const { itemId } of items) {
      try { await ctx.runAction(internal.priceWatch.checkItem, { itemId }); }

export const checkItem = internalAction({
  args: { itemId: v.id("items") },
  handler: async (ctx, { itemId }) => {
    const item = await ctx.runQuery(internal.priceWatch.getItem, { itemId });
    if (!item?.productUrl) return;
    const doc = await firecrawl.scrape(ctx, item.productUrl, {
      formats: ["markdown", { type: "json", prompt: `Return the current selling price for "${item.name}" as {"price": number, "currency": "ISO code"}. Use the sale price if one is shown. If unavailable, price must be null.` }],
      onlyMainContent: true,
      maxAge: 3_600_000,
    });
    let price: number | null = null;
    let currency: string | null = null;
    const j = doc.json as any;
    if (j && typeof j.price === "number") { price = j.price; currency = j.currency ?? null; }
    if (price === null && doc.markdown) {
      const p = await extract("price", Price, `Extract the current price of the product named "${item.name}" from this page text. Use the sale price if shown.`, doc.markdown);
      price = p.price; currency = p.currency;
    }
    await ctx.runMutation(internal.priceWatch.recordCheck, {
      itemId, observedCents: price === null ? undefined : toCents(price), sourceUrl: item.productUrl,
      note: currency ? `currency ${currency}` : undefined,
    });
  },
});

  handler: async (ctx, args) => {
    const item = await ctx.db.get(args.itemId);
    if (!item) return;

    if (!policy?.windowDays) return;
    const ends = windowEndsAt(purchase.purchasedAt, policy.windowDays);
    if (ends < Date.now()) return;
    const drop = priceDropCents(item.unitCents, args.observedCents, item.qty);
    if (drop === null) return;

  },
});

    const prev = await ctx.db.query("drafts").withIndex("by_claim", (q) => q.eq("claimId", args.claimId)).collect();
    const id = await ctx.db.insert("drafts", { ...args, version: prev.length + 1, claimVersion: claim.version });

export const approveAndSend = mutation({
  args: { draftId: v.id("drafts"), to: v.string(), subject: v.string(), body: v.string() },
  handler: async (ctx, args) => {
    const userId = await requireUserId(ctx);
    const draft = await ctx.db.get(args.draftId);
    if (!draft || draft.userId !== userId) throw new ConvexError("Draft not found");
    const claim = await ownedClaim(ctx, draft.claimId, userId);
    if (claim.version !== draft.claimVersion) throw new ConvexError("The claim changed since this draft was written. Generate a new draft.");
    if (draft.outboundId) throw new ConvexError("This draft was already sent");
    const to = args.to.trim();
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(to)) throw new ConvexError("Enter a valid recipient email");
    const profile = await ctx.db.query("profiles").withIndex("by_user", (q) => q.eq("userId", userId)).unique();
    if (!profile) throw new ConvexError("Set up your Recoup inbox first");
    const subject = args.subject.includes(`[RC-${claim.token}]`) ? args.subject : `${args.subject} [RC-${claim.token}]`;
    const outboundId = await agentmail.sendMessage(ctx, profile.inboxId, {
      to, subject, text: args.body, labels: [`claim:${claim._id}`],
    });
    await ctx.db.patch(draft._id, { to, subject, body: args.body, approvedAt: Date.now(), outboundId });
    await ctx.db.patch(claim._id, { status: "sent", version: claim.version + 1, attentionAt: undefined });
    const fresh = (await ctx.db.get(claim._id))!;
    await scheduleReminder(ctx, fresh, Date.now() + 7 * 86_400_000);
    await ctx.scheduler.runAfter(30_000, internal.drafts.captureThread, { draftId: draft._id, attempt: 1 });
    return outboundId;
  },
});

export const captureThread = internalMutation({
  args: { draftId: v.id("drafts"), attempt: v.number() },
  handler: async (ctx, { draftId, attempt }) => {

    if (!draft?.outboundId) return;

      const claim = await ctx.db.get(draft.claimId);
      if (claim && !claim.threadId) await ctx.db.patch(claim._id, { threadId: status.threadId });
      return;
    }
    if (attempt < 5) await ctx.scheduler.runAfter(60_000 * attempt, internal.drafts.captureThread, { draftId, attempt: attempt + 1 });
  },
});

"credit_issued" = they say it has already been issued. "refusal" = they decline. "question" = they need more information. Otherwise "other". promisedAmount is the amount they mention, or null.`;

      const unresolved = claim.expectedCents - events.filter((e) => e.kind === "confirmed_credit").reduce((s, e) => s + e.cents, 0) + events.filter((e) => e.kind === "later_debit").reduce((s, e) => s + e.cents, 0);
      const cents = args.promisedAmount !== undefined ? toCents(args.promisedAmount) : Math.max(unresolved, 0);
